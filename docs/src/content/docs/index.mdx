## Automate in code, not YAML

```ts title="setup.ts"
import { apply } from 'sysopkit';
import { start } from 'sysopkit/start';
import { SSHConnector } from 'sysopkit/connector/ssh';
import { sudo } from 'sysopkit/middleware/sudo';
import { createFile } from 'sysopkit/op/file';
import { sh } from 'sysopkit/op/sh';

await start(async () => {
  await using c = new SSHConnector({ host: '192.168.1.10', user: 'sysop' });

  await apply('setup nginx', c, async () => {
    await sudo(async () => {
      await sh('apt install -y nginx');
      await createFile({
        path: '/etc/nginx/sites-available/default',
        content: 'server { listen 80; }',
        mode: 0o644,
      });
    });
  });
});
```

Run it with `bun setup.ts`. The script connects over SSH, installs nginx with elevated privileges, and writes the config file idempotently — re-running it changes nothing. No agents, no DSL: just TypeScript with compile-time checks, `try/catch` error handling, and npm modules for reuse.
